-- =====================================
-- CUSTOMER CHURN ANALYSIS
-- =====================================


--1. See column structure
PRAGMA table_info(customers);

--2. Preview actual datacustomers
SELECT * FROM customers
LIMIT 10;

--3. Check total customers
SELECT COUNT(*) AS total_customers
FROM customers;

--4. Check churn distribution
SELECT churn_label, COUNT(*) AS customers
FROM customers
GROUP BY churn_label;

--5. Overall Churn Rate
SELECT
COUNT(*) AS total_customers,
SUM(CASE WHEN churn_label='Yes' THEN 1 ELSE 0 END) AS churned_customers,
ROUND(
SUM(CASE WHEN churn_label='Yes' THEN 1 ELSE 0 END)*100.0/COUNT(*),2
) AS churn_rate
FROM customers;

--6. Find WHICH customers churn more(Which contract type has highest churn?)
SELECT 
contract,
COUNT(*) AS total_customers,
SUM(CASE WHEN churn_label = 'Yes' THEN 1 ELSE 0 END) AS churned_customers,
ROUND(
    SUM(CASE WHEN churn_label = 'Yes' THEN 1 ELSE 0 END) * 100.0 / COUNT(*),
2
) AS churn_rate
FROM customers
GROUP BY contract
ORDER BY churn_rate DESC;

--7. Find WHY customers churn (Internet Service Analysis)
SELECT 
internet_service,
COUNT(*) AS total_customers,
SUM(CASE WHEN churn_label = 'Yes' THEN 1 ELSE 0 END) AS churned_customers,
ROUND(
    SUM(CASE WHEN churn_label = 'Yes' THEN 1 ELSE 0 END) * 100.0 / COUNT(*),
2
) AS churn_rate
FROM customers
GROUP BY internet_service
ORDER BY churn_rate DESC;

--8. Find the MOST IMPORTANT churn driver (Monthly Charges)
SELECT 
CASE 
    WHEN monthly_charges < 35 THEN 'Low'
    WHEN monthly_charges BETWEEN 35 AND 70 THEN 'Medium'
    ELSE 'High'
END AS price_category,
COUNT(*) AS total_customers,
SUM(CASE WHEN churn_label = 'Yes' THEN 1 ELSE 0 END) AS churned_customers,
ROUND(
    SUM(CASE WHEN churn_label = 'Yes' THEN 1 ELSE 0 END) * 100.0 / COUNT(*),
2
) AS churn_rate
FROM customers
GROUP BY price_category
ORDER BY churn_rate DESC;

--9. Highest churn rate
SELECT 
    CASE 
        WHEN tenure_months <= 12 THEN '0-1 Year'
        WHEN tenure_months <= 24 THEN '1-2 Years'
        WHEN tenure_months <= 48 THEN '2-4 Years'
        ELSE '4+ Years'
    END AS tenure_group,
    
    COUNT(*) AS total_customers,
    
    SUM(CASE WHEN churn_label = 'Yes' THEN 1 ELSE 0 END) AS churned_customers,
    
    ROUND(
        SUM(CASE WHEN churn_label = 'Yes' THEN 1 ELSE 0 END) * 100.0 
        / COUNT(*), 2
    ) AS churn_rate

FROM customers
GROUP BY tenure_group
ORDER BY churn_rate DESC;

--10. Churn by Tenure Group
SELECT
    CASE
        WHEN tenure_months <= 12 THEN '0-1 Year'
        WHEN tenure_months <= 24 THEN '1-2 Years'
        WHEN tenure_months <= 48 THEN '2-4 Years'
        ELSE '4+ Years'
    END AS tenure_group,
    COUNT(*) AS total_customers
FROM customers
GROUP BY tenure_group
ORDER BY total_customers DESC;



--11. Churn by Price Category
SELECT
    CASE
        WHEN monthly_charges < 35 THEN 'Low Price'
        WHEN monthly_charges BETWEEN 35 AND 70 THEN 'Medium Price'
        ELSE 'High Price'
    END AS price_category,
    COUNT(*) AS total_customers
FROM customers
GROUP BY price_category;

--12. full churn query
SELECT
    CASE
        WHEN monthly_charges < 35 THEN 'Low Price'
        WHEN monthly_charges BETWEEN 35 AND 70 THEN 'Medium Price'
        ELSE 'High Price'
    END AS price_category,

    COUNT(*) AS total_customers,

    SUM(CASE 
        WHEN churn_label = 'Yes' THEN 1 
        ELSE 0 
    END) AS churned_customers,

    ROUND(
        SUM(CASE WHEN churn_label = 'Yes' THEN 1 ELSE 0 END) * 100.0 
        / COUNT(*),
    2) AS churn_rate

FROM customers
GROUP BY price_category;

--13. Identify high risk customers
SELECT
    contract,

    CASE 
        WHEN tenure_months <= 12 THEN 'New Customers'
        WHEN tenure_months <= 36 THEN 'Mid-Term Customers'
        ELSE 'Long-Term Customers'
    END AS customer_age_group,

    CASE
        WHEN monthly_charges < 35 THEN 'Low Price'
        WHEN monthly_charges BETWEEN 35 AND 70 THEN 'Medium Price'
        ELSE 'High Price'
    END AS price_category,

    COUNT(*) AS total_customers,

    SUM(CASE WHEN churn_label = 'Yes' THEN 1 ELSE 0 END) AS churned_customers,

    ROUND(
        SUM(CASE WHEN churn_label = 'Yes' THEN 1 ELSE 0 END) * 100.0
        / COUNT(*),
    2) AS churn_rate

FROM customers
GROUP BY contract, customer_age_group, price_category
HAVING total_customers > 50
ORDER BY churn_rate DESC;
